# GET ('/entries)
- read all journal entries

# GET ('/entries/new)
- shows the empty add form 

# POST ('/entries')
- create a new journal entry

# GET ('/entries/:id')
- read one entry

# GET ('/entries/edit/:id')
- read filled entry 

# PUT ('/entries/:id')
-  update filled entry (fetch)

# DELETE ('/entries/:id')
- delete a filled entry (fetch)
